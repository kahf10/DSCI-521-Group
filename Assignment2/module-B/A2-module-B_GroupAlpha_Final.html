import gender_guesser.detector as gender

def add_gender_column(df):
    
    #--- Your code starts here

    # Initialize the gender detector
    detector = gender.Detector(case_sensitive=False)

    def map_gender(name):
        result = detector.get_gender(name)
        if result in ['male', 'mostly_male']:
            return 'male'
        elif result in ['female', 'mostly_female']:
            return 'female'
        else:
            return 'unknown'

    # Apply mapping function
    df['gender'] = df['first_name'].apply(map_gender)
    df = df[df['gender'] != 'unknown'] # Retain only employees with classified gender
    
    #--- Your code ends here
    
    return df

import matplotlib.pyplot as plt

def barplot(df, column, title, top = 5):
    
    ## It's width by height, so let's make this one wide!
    fig = plt.figure(figsize = (8, 4))

    #--- Your code starts here

    # Get the counts for the top categories
    top_categories = df[column].value_counts().head(top)

    # Plot
    plt.bar(top_categories.index, top_categories.values)
    
    #--- Your code ends here

    ## Set the tick and label fontsize
    plt.tick_params(labelsize = 15)

    ## Set the title
    _ = plt.title(title, fontsize = 15)

    ## Set the x,y-labels
    _ = plt.xticks(rotation = 'vertical', fontsize = 13)
    _ = plt.ylabel("Number of employees", fontsize = 13)
    
    return fig

def add_overtime(df):

    #--- Your code starts here

    # Calculate hourly rate
    df['hourly_rate'] = df['annual_salary'] / 2087

    # Calculate overtime hours worked
    df['ytd_overtime_gross_hours'] = df['ytd_overtime_gross'] / (1.5 * df['hourly_rate'])

    # Remove NaN and infinite values
    df['ytd_overtime_gross_hours'].replace([np.inf, -np.inf], np.nan)
    df.dropna(subset=['ytd_overtime_gross_hours'])
    #--- Your code ends here
    
    return df
